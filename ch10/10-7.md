# 10.7 モナドなど

* 「モナド」とは、プログラミングパターンの1つで、「モナディック」な関数を直列につなげることができるもの
  * 「モナディック」な関数とは、「通常」の値を受け取り、ある種の「拡張」された値を返す関数
      * 本書でこれまで扱ってきた例としては、拡張された値＝Result型でラップされた値
* 「モナド」は、技術的には次の3要素を持ったもの
  1. データ構造
  2. 関連するいくつかの関数（returnとbind）
  3. 関数がどのように動作しなければいけないかについてのいくつかのルール（＝モナド則）

```fs
// --------------------------------------------------------
// 1. データ構造
// --------------------------------------------------------
type Result<'Success, 'Failure> =
  | Ok of 'Success
  | Error of 'Failure

// --------------------------------------------------------
// 2. returnとbind
// --------------------------------------------------------
// returnは、通常の値をモナドに包む関数
let ret x = Ok x

// bindは、モナドの値にモナディック関数をつないでいける関数
let bind m f =
    match m with
    | Ok x -> f x
    | Error e -> Error e

// --------------------------------------------------------
// 3. モナド則
// --------------------------------------------------------
// 3.1. 左単位元：通常の値をモナドに包んでから f に渡すのは、最初から f に渡すのと同じ
//      bind (ret x) f == f x
let x = 20
let f x = if x % 2 = 0 then Ok (x / 2) else Error "even"
let left = bind (ret x) f // => Ok 10
let right = f x           // => Ok 10

// 3.2. 右単位元：モナドの値にreturnを適用しても、何も変わらない
//      bind m ret ==  m
let m = Ok 10
let left = bind m ret // => Ok 10
let right = m         // => Ok 10

// 3.3. 結合則：つなげるのも入れ子にするのも同じ、…と雑に理解
//      bind (bind m f) g  ==  bind m (fun x -> bind (f x) g)
let left = bind (bind m f) f  // => Ok 5
let right = bind (bind m f) f  // => Ok 5
```

## 10.7.1 アプリカティブを使って並列に合成する

* アプリカティブを使うと、モナディック関数を直列につなげるのではなく、並列に組み合わせることが可能になる
* モナディックなアプローチでは、バリデーションエラーが発生した際に最初のエラーしか残せないが、アプリカティブなアプローチではすべてのエラーを結合できるようになる

```fs
// --------------------------------------------------------
// アプリカティブでバリデーションを扱う構造
// --------------------------------------------------------
type Validation<'T> = Result<'T, string list>

let pure x : Validation<'T> = Ok x

let apply fRes xRes : Validation<'T> =
    match fRes, xRes with
    | Ok f, Ok x -> Ok (f x)
    | Error e1, Error e2 -> Error (e1 @ e2)  // エラーメッセージを蓄積していく
    | Error e, _ -> Error e
    | _, Error e -> Error e

// --------------------------------------------------------
// 個々のバリデーション関数
// --------------------------------------------------------
let validateName name =
    if name = "" then Error ["名前が空です"] else Ok name

let validateAge age =
    if age < 0 then Error ["年齢がマイナスです"]
    elif age > 150 then Error ["年齢が高すぎます"]
    else Ok age

let validateEmail email =
    if email.Contains("@") then Ok email
    else Error ["メールアドレスが無効です"]

// --------------------------------------------------------
// バリデーションの合成
// --------------------------------------------------------
type User = {
    Name: string
    Age: int
    Email: string
}
let createUserApplicative name age email =
    pure (fun n a e -> { Name = n; Age = a; Email = e })
    |> apply (validateName name)  // ここでバリデーションエラーになっても次のバリデーションに進んでいく
    |> apply (validateAge age)
    |> apply (validateEmail email)
```
